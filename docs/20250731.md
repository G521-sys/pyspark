-- 1. 启用数据库 CDC（如果尚未启用）
USE page;
GO

IF (SELECT is_cdc_enabled FROM sys.databases WHERE name = 'page') = 0
BEGIN
    EXEC sys.sp_cdc_enable_db;
    PRINT 'CDC enabled for database page';
END
ELSE
BEGIN
    PRINT 'CDC already enabled for database page';
END
GO

-- 2. 启用表 CDC
IF (SELECT is_tracked_by_cdc FROM sys.tables WHERE name = 'page_log') = 0
BEGIN
    EXEC sys.sp_cdc_enable_table
        @source_schema = N'dbo',
        @source_name   = N'page_log',
        @role_name     = NULL;
    PRINT 'CDC enabled for table dbo.page_log';
END
ELSE
BEGIN
    PRINT 'CDC already enabled for table dbo.page_log';
END
GO

-- 3. 验证
SELECT
    d.name AS database_name,
    d.is_cdc_enabled,
    t.name AS table_name,
    t.is_tracked_by_cdc
FROM sys.databases d
JOIN sys.tables t ON 1=1
WHERE d.name = 'page' AND t.name = 'page_log';
