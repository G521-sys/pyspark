WITH sales_monthly AS (
    -- 第一步：按区域、产品、年月聚合，计算每月销售额
    SELECT
        region,
        product,
        DATE_FORMAT(order_date, 'yyyy-MM') AS sales_month, -- 提取年月
        SUM(amount) AS monthly_sales -- 当月销售额
    FROM sales_order
    GROUP BY
        region,
        product,
        DATE_FORMAT(order_date, 'yyyy-MM')
)

SELECT
    region,
    product,
    sales_month,
    monthly_sales AS current_sales, -- 当月销售额
    -- 第二步：用 LAG 开窗函数，根据年月偏移 12 行（去年同期），获取去年同期销售额
    LAG(monthly_sales, 12) OVER (
        PARTITION BY region, product
        ORDER BY sales_month
    ) AS last_year_sales,
    -- 第三步：计算同比增长金额（当月 - 去年同期）
    monthly_sales - LAG(monthly_sales, 12) OVER (
        PARTITION BY region, product
        ORDER BY sales_month
    ) AS growth_amount,
    -- 第四步：计算同比增长率（增长金额 / 去年同期销售额 * 100 ），保留 2 位小数
    ROUND(
        (monthly_sales - LAG(monthly_sales, 12) OVER (
            PARTITION BY region, product
            ORDER BY sales_month
        ))
        / NULLIF(LAG(monthly_sales, 12) OVER (
            PARTITION BY region, product
            ORDER BY sales_month
        ), 0) * 100,
        2
    ) AS growth_rate
FROM sales_monthly
-- 第五步：按区域、产品分组，年月排序（开窗函数已指定排序，此处可增强语义）
ORDER BY
    region,
    product,
    sales_month;
